* Installation
** Use Package
#+BEGIN_SRC sh
git clone https://github.com/tychoish/xtdlib.el ~/path/to/xtdlib.el
#+END_SRC

#+BEGIN_SRC emacs-lisp
(use-package xtdlib
  :load-path "~/path/to/xtdlib.el"
  :demand t)
#+END_SRC

#+RESULTS:

This works with any Emacs version that has =use-package= (28+) and does not require network access at startup.

* API Reference
#+BEGIN_SRC emacs-lisp :results output raw :exports none
  (save-excursion
    (org-back-to-heading t)
    (let ((end (save-excursion (org-end-of-subtree t) (point))))
      (forward-line 1)
      (while (and (< (point) end) (not (org-at-heading-p)))
        (forward-line 1))
      (when (< (point) end)
        (delete-region (point) end))))

  (let* ((org-dir (file-name-directory (buffer-file-name)))
         (el-file (car (seq-remove
                        (lambda (f) (string-match-p "-test\\.el$\\|test-.*\\.el$" f))
                        (directory-files org-dir t "\\.el$" t)))))
    (add-to-list 'load-path org-dir)
    (require (intern (file-name-base el-file)))

    (defun xtdlib-doc--section-name (comment-text)
      (let* ((name (string-trim (car (split-string comment-text " -- "))))
             (name (replace-regexp-in-string "-" " " name)))
        (mapconcat (lambda (s) (if (string-match-p "\\." s) s (capitalize s)))
                   (split-string name "[ \t]+" t) " ")))

    (defun xtdlib-doc--format-doc (doc)
      (let* ((paras (split-string doc "\n\n" t))
             (fn-para (seq-find (lambda (p) (string-match "\\`(fn " (string-trim p))) paras))
             (rest-paras (seq-remove (lambda (p) (string-match "\\`(fn " (string-trim p))) paras)))
        (mapconcat #'identity
                   (delq nil
                         (list
                          (when fn-para (concat ": " (string-trim fn-para)))
                          (when rest-paras
                            (mapconcat
                             (lambda (para)
                               (if (string-match "\\`[ \t]" para)
                                   para
                                 (mapconcat #'string-trim (split-string para "\n" t) " ")))
                             rest-paras "\n\n"))))
                   "\n\n")))

    (defun xtdlib-doc--format-sym (name)
      (let* ((sym (intern name))
             (fn-p (fboundp sym))
             (var-p (and (not fn-p) (boundp sym)))
             (doc (or (when fn-p (documentation sym))
                      (when var-p
                        (documentation-property sym 'variable-documentation)))))
        (format "*** %s~%s~\n\n%s\n\n"
                (if doc "" "TODO ")
                name
                (if doc (xtdlib-doc--format-doc doc) "*no docstring*"))))

    ;; Parse sections from the el file by scanning for ruler/comment patterns.
    ;; Sections are delimited by long ;;;;... ruler lines; the first ;; comment
    ;; after each ruler names the section.
    (let (sections current-section current-syms after-ruler)
      (with-temp-buffer
        (insert-file-contents el-file)
        (goto-char (point-min))
        (while (not (eobp))
          (let ((line (buffer-substring-no-properties
                       (line-beginning-position) (line-end-position))))
            (cond
             ((string-match "^;\\{10,\\}" line)
              (when current-section
                (push (cons current-section (nreverse current-syms)) sections))
              (setq current-section nil current-syms nil after-ruler t))
             ((and after-ruler
                   (string-match "^;; \\(.+\\)" line))
              (setq current-section (xtdlib-doc--section-name (match-string 1 line))
                    after-ruler nil))
             ((and current-section
                   (string-match "^(\\(?:cl-\\)?def[^ ]+ +'?\\([^ ()]+\\)" line))
              (push (match-string 1 line) current-syms))))
          (forward-line 1))
        (when current-section
          (push (cons current-section (nreverse current-syms)) sections)))

      (dolist (section (nreverse sections))
        (princ (format "** %s\n\n" (car section)))
        (dolist (name (cdr section))
          (princ (xtdlib-doc--format-sym name))))))
#+END_SRC

** Slow Op

*** ~slow-op-reporting~

A toggle that, when enabled is supports more verbose timing reporting. Turns ‘with-slow-op-timer’ from a noop to reporting on the duration of enclosed operations.

*** ~slow-op-threshold~

Threshold in seconds, or fractions thereof. Controls the behavior of ‘with-slow-op-timer’. Any operation below this threshold (faster) are ignored. Use this to control verbosity.

*** ~with-slow-op-timer~

Send a message the BODY operation of NAME takes longer to execute than a hardcoded threshold.

** Strings

*** ~s-equal~

: (fn S1 S2)

Return t if two strings have identical contents. Case is significant, but text properties are ignored. Symbols are also allowed; their print names are used instead.

See also ‘string-equal-ignore-case’.

*** ~s-empty-p~

: (fn STRING)

Check whether STRING is empty.

*** ~-filter-s-trim~

Return STRS with non-strings removed and remaining strings trimmed of whitespace. Empty strings and strings that are entirely whitespace are excluded from the result.

*** ~s-or-char-equal~

Return t when CHAR equals VALUE, where VALUE may be a character or single-char string.

*** ~s-define-join-string-function~

: (fn CHAR &optional &key USE-JARGON-NAMES SPACE-PADDING)

Define a function ‘s-join-with-NAME’ that joins non-empty strings with CHAR as separator. NAME is derived from CHAR via ‘xtdlib--join-char-names’, falling back to the Unicode character name. When USE-JARGON-NAMES is non-nil, the alternate jargon name is used (e.g. kebab vs hyphen). When SPACE-PADDING is non-nil, the separator is surrounded by spaces.

*** ~s-shortest~

Return the shorter of strings A and B. When equal in length, return A.

*** ~s-collapse-hyphens~

Replace runs of three or more consecutive hyphens in STR with a single hyphen.

*** ~s-normalize-symbol-name~

Normalize NAME to a clean hyphen-separated string suitable for use as a symbol name. Trims outer whitespace, collapses internal whitespace, replaces common punctuation with hyphens, and collapses runs of three or more hyphens.

*** ~s-trimmed-or-nil~

Return VALUE trimmed of surrounding whitespace, or nil if VALUE is not a string or is empty after trimming.

*** ~s-trim-non-word-chars~

Trim leading and trailing non-word characters from VALUE. Return nil if VALUE is not a string or is empty after trimming.

*** ~s-blank-p~

Return t when VALUE is a string that is empty or consists entirely of whitespace. Returns nil for non-strings, non-empty strings, and strings with non-whitespace content. Use this to guard against blank user input or empty configuration values.

*** ~s-contains-whitespace-p~

Return t when VALUE is a string that is empty or consists entirely of whitespace. Returns nil for non-strings, non-empty strings, and strings with non-whitespace content. Use this to guard against blank user input or empty configuration values.

*** ~s-default~

Return DEFAULT when INPUT is nil or an empty string, otherwise return INPUT.

** Lists

*** ~larger~

Return the larger value. If either FIRST or SECOND are not numbers, treat them as 0.

*** ~smaller~

Return the smaller value. If either FIRST or SECOND are not numbers, treat them as 0.

*** ~make-add-to-list-fn~

: (fn LIST &optional &key APPEND)

Return a lambda that appends ITEM to LIST via ‘add-to-list’. LIST may be a bare symbol or a quoted symbol.

** `dash.el'

*** ~-distinct-by-car~

: (fn CELL &optional &key (TEST #'equal))

Take a list of cons cells and return a new list that contains only the elements that have unique car values, ignoring the cdr entirely. Compare values using the test function, which defaults to ‘equal’.

*** ~-distinct-paths~

Return CELL with duplicate file paths removed, using ‘f-equal-p’ for comparison. Unlike ‘-distinct’, this handles paths that differ only in trailing slashes or relative vs. absolute form when the filesystem resolves them to the same location.

*** ~-distinct-by-alist-key~

: (fn KEY CELL &optional &key (TEST #'equal))

Compare a list of alists, and return a new list that contains only the alists that have distinct values for a specific key. Compare values using the test function, which defaults to ‘equal’.

*** ~-unwind~

Flatten LIST by exactly one level, collecting nested list items into a single list. Unlike ‘-flatten’, which recurses fully, ‘-unwind’ only removes one layer of nesting.

*** ~-flat-map~

‘-flat-map’ applies (maps) a function, which returns a list, to each item in a list. The lists that result from the map operation are then concatenated or joined. This provides a dash.el conforming API for the ‘mapcan’ operation.

*** ~-mapc~

: (fn FUNCTION SEQUENCE)

Apply FUNCTION to each element of SEQUENCE for side effects only. Unlike ‘mapcar’, don’t accumulate the results.  Return SEQUENCE. SEQUENCE may be a list, a vector, a bool-vector, or a string.

*** ~-join~

: (fn &rest LISTS)

Concatenate any number of lists by altering them. Only the last argument is not altered, and need not be a list.

*** ~-append~

: (fn &rest SEQUENCES)

Concatenate all the arguments and make the result a list. The result is a list whose elements are the elements of all the arguments. Each argument may be a list, vector or string.

All arguments except the last argument are copied.  The last argument is just used as the tail of the new list.  If the last argument is not a list, this results in a dotted list.

As an exception, if all the arguments except the last are nil, and the last argument is not a list, the return value is that last argument unaltered, not a list.

*** ~-reverse~

: (fn SEQ)

Reverse order of items in a list, vector or string SEQ. If SEQ is a list, it should be nil-terminated. This function may destructively modify SEQ to produce the value.

*** ~-sparse~

: (fn LIST)

Return a copy of LIST with all nil items removed.

*** ~-c~

: (fn CAR CDR)

Create a new cons, give it CAR and CDR as components, and return it.

*** ~-l~

: (fn &rest OBJECTS)

Return a newly created list with specified arguments as elements. Allows any number of arguments, including zero.

*** ~-strings~

Return INPUT as a flat list, optionally coercing non-string elements. A trailing ‘:options OPT’ pair in INPUT selects how non-strings are handled: ‘filter’ drops them, ‘stringify’ formats them with ‘format’. OPT may be a list combining these symbols. When ‘:options’ is not provided or its value is nil, INPUT is returned as-is without coercion. A ‘user-error’ is signaled only when OPT is some other, unrecognized value.

*** ~-sparse-append~

Append ITEMS and remove all nil values from the concatenated result.

*** ~-map-in-place~

Apply the ‘mapper’ function to every item in the list ‘items’ and replace the items in the original list with the results of the function, returning the list. This is a destructive operation.

*** ~-in-place~

Destructively apply MAPPER to every item in ITEMS, replacing each element in place. Returns the count of elements processed. Use this when you need the element count rather than the list itself; prefer ‘-map-in-place’ when you need the modified list back.

*** ~-map-uniq~

: (fn MAPPER INPUT &optional &key (TEST #'equal))

Apply the ‘mapper’ function to every item in the ‘input’ list, returning a new list that contains the unique output of the list. Comparisons use the ‘test’ function, which defaults to ‘equal’.

*** ~--mapc~

Apply the form (with the current element avalible as the variable ‘it’) to all item in the list, primarily for side effects. Returns the input list. This is an anaphoric equivalent to ‘mapc’. As opposed to ‘--each’ and ‘-each’, which return nil, ‘-mapc’ returns the input list.

This is the anaphoric counterpart to ‘-mapc’.

*** ~--flat-map~

‘--flat-map’ evaluates a form for very item in ‘input-list’ with the item bound to ‘it’. The form must return a list, and the returned lists are then concatenated or joined into a single flattened list. This provides a dash.el conforming API for the ‘mapcan’ operation.

This is the anaphoric counterpart to ‘-flat-map’.

*** ~--map-in-place~

Apply the form, (with the current element as ‘it’) to each item in the list, distructively setting the return value of the form to the value in the list.

This is the anaphoric counterpart to ‘-map-in-place’.

*** ~--in-place~

Take a list and replace each element in the list with the result of evaluating ‘FORM’ for that element. The original element is accessable in the ‘FORM’ as ‘it‘. Returns the number of items in the list. This is a destructive operation.

This is the anaphoric counterpart to ‘-in-place’.

*** ~--map-uniq~

Apply ‘FORM’ to every item in the input list, available as ‘it’, and collect the UNIQUE results, and returning them as a list. Items are compared for uniqueness with ‘equal’ by default, which can be overridden with the ‘-compare-fn’ dynamic variable.

This is the anaphoric counterpart to -map-unique. Although the handling of the equality function customization differs slightly.

** `f.el'

*** ~f-mtime~

Return the modification time of FILENAME as a time value.

*** ~f-atime~

Return the access time of FILENAME as a time value.

*** ~f-make-slug~

Turn a string, S, into a hyphen-seperated slug for filename.

*** ~f-filename-is-p~

Return non-nil if the filename component of path ENTRY equals NAME.

*** ~f-when-file-exists~

Return PATH if it exists on the filesystem, otherwise nil.

*** ~f-distinct~

Return SEQUENCE with duplicate paths removed using ‘f-equal-p’ for comparison.

*** ~f-directories-containing-file-with-extension-function~

Define ‘f-EXT-file-p’ and ‘f-directories-containing-file-with-extension-EXT’ for EXTENSION.

*** ~f-files-in-directory~

Return a flat list of all files under PATH. PATH may be a directory, a file (returns siblings), or a list of paths.

*** ~f-recursive-directories-containing~

Return a list of directories under PATH that contain a file named FILENAME. Searches recursively. PATH defaults to ‘default-directory’ when nil.

*** ~f-directories-containing-file-function~

Define helper predicates and a search function for directories containing FILENAME. Also accepts additional FILES as alternate names to match.

*** ~f-collapse-homedir~

Replace the expanded home directory prefix in PATH with ‘~/’.

*** ~f-visually-compress-path~

Truncate each component of PATH to at most NUM characters, preserving the separator.

*** ~f-visual-compression-function~

Define ‘f-visually-compress-to-NAME’ that truncates each path component to NUM characters.

*** ~f-filter-directories~

Return entries from SEQUENCE that are existing directories. OPTIONS is a symbol or list: ‘cannonicalize’ expands paths to absolute form, ‘unique’ deduplicates results via ‘f-equal-p’.

** Utility Macros

*** ~disabled~

Wrap BODY so it never executes. Keeps code byte-compilable while effectively commenting it out.

*** ~with-force-write~

Temporarily clear ‘buffer-read-only’, evaluate BODY, then restore it to t.

*** ~pos-arg~

Allow positional arguments to have annotated call-sites.

*** ~pa~

Allow positional arguments to have annotated call-sites.

*** ~add-one-shot-hook~

: (fn &key NAME HOOK FUNCTION RESULT BODY FORM OPERATION (ARGS nil) (LOCAL nil) (PERSIST nil) (COUNT 1) (DEPTH 0) (MAKE-UNIQUE nil) (CLEANUP nil) (IDLE-TIMER nil))

Register a self-removing hook function named NAME on HOOK. The hook body is specified via one of: FUNCTION (a symbol or lambda), FORM (an expression), BODY (a list of forms), RESULT (evaluated at expansion time), or OPERATION (called via funcall/apply with optional ARGS).

After firing COUNT times (default 1) the hook removes itself. Set PERSIST to t to keep the hook active indefinitely. When IDLE-TIMER is a number, the body runs in an idle timer of that many seconds rather than directly in the hook.

HOOK may be a symbol, a list of symbols, or the special sentinel ‘after-first-frame-created’, which routes to ‘window-setup-hook’ in non-daemon sessions and ‘server-after-make-frame-hook’ in daemon sessions.

DEPTH controls hook insertion depth (default 0). LOCAL makes the hook buffer-local. MAKE-UNIQUE generates an uninterned symbol to avoid name collisions. CLEANUP uninterns the generated symbol after the hook fires.

*** ~make-run-hooks-function-for~

Define a zero-argument function ‘run-hooks-for-MODE’ that runs ‘MODE-hook’.

*** ~create-toggle-functions~

: (fn VALUE &optional &key SHORT-NAME LOCAL KEYMAP KEY)

Define turn-on, turn-off, and toggle interactive commands for variable VALUE. Use SHORT-NAME to override the generated name. LOCAL makes commands use ‘setq-local’. Optionally bind the toggle to KEY in KEYMAP.

*** ~make-read-extended-command-for-prefix~

: (fn PREFIX &optional &key BIND-MAP BIND-KEY KEY-ALIAS)

Define an interactive command that runs ‘execute-extended-command’ filtered to PREFIX. Only commands whose names begin with PREFIX are offered for completion. Optionally bind the command to BIND-KEY in BIND-MAP with KEY-ALIAS as the which-key label.

*** ~with-toggle-once~

Define a function NAME that executes BODY only the first time it is called. Subsequent calls are no-ops. Uses an auto-generated toggle variable to guard execution.

*** ~with-prefix-arg~

Evaluate BODY with ‘current-prefix-arg’ bound to ARG.

*** ~with-default-directory~

Run the body with ‘default-directory’ set to the path provided

*** ~with-silence~

Totally suppress message from either the minibuffer or the *Messages* buffer..

*** ~with-quiet~

Suppress any messages from appearing in the minibuffer area.

*** ~with-temp-keymap~

Bind MAP to a fresh sparse keymap, evaluate BODY, and return the keymap. Use this to build a keymap programmatically and return it in one expression.

*** ~setq-when-nil~

: (fn VARIABLE VALUE &optional &key LOCAL)

Set VARIABLE to VALUE only if VARIABLE is currently nil. Use :local t for ‘setq-local’.

*** ~with-timer~

Report on NAME and the time taken to execute BODY.

*** ~compile-buffer-name~

Return a function suitable for ‘compilation-buffer-name-function’ that always returns NAME. The returned function accepts one optional argument (the compilation mode) and ignores it.

*** ~merge-predicate-functions~

Return a lambda that applies each predicate in PREDS to its argument with ‘and’. Short-circuits on the first predicate that returns nil, consistent with ‘and’ semantics.

** Project Context

*** ~buffer-derived-mode-p~

Returns t when the major-mode of ‘buffer’ is or is derived from ‘mode.’

*** ~mode-buffers-for-project~

: (fn &optional &key (MODE major-mode))

Return buffers in the current project whose major mode derives from MODE. MODE defaults to ‘major-mode’ of the calling buffer. Uses ‘approximate-project-buffers’ to determine project membership.

*** ~mode-buffers~

: (fn &optional (MODE major-mode))

Return all live buffers whose major mode derives from MODE. MODE defaults to ‘major-mode’ of the calling buffer. Searches across all buffers, not just the current project.

*** ~package-available-p~

Return non-nil if the feature NAME is currently loaded via ‘featurep’. Use this to guard optional integrations with packages that may not be present.

*** ~approximate-project-root~

Return the current project root, falling back to ‘default-directory’.

*** ~approximate-project-name~

Return the current project name, falling back to the directory basename.

*** ~approximate-project-buffers~

Return buffers belonging to the current project.
